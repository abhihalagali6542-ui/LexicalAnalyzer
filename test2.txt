void read_array(int *arr,int size)
{
        for(int i=0;i<size;i++)
                scanf("%d",&arr[i]);
}

int marge_arr(int *arr1,int *arr2,int size1,int size2,int *arr3)
{
        int i=0,j=0;
        while(i<size1)
        {
                arr3[i]=arr1[i];
                i++;
        }
        while(j<size2)
        {
                arr3[i++]=arr2[j++];
        }
        return i;
}

void bubble_sort(int *arr,int size)
{
        for(int i=0;i<size-1;i++)
        {
                for(int j=0;j<size-1-i;j++)
                {
                        if(arr[j]>arr[j+1])
                        {
                                int temp=arr[j];
                                arr[j]=arr[j+1];
                                arr[j+1]=temp;
                        }
                }
        }
}

int find_median(int *arr,int size)
{
        if(size%2==0)
        {
                return (arr[size/2]+arr[size/2-1])/2;
        }
                return arr[size/2];
}


int main()
{
        int size1,size2;
        printf("Enter two size: ");
        scanf("%d %d",&size1,&size2);
        int arr1[size1],arr2[size2];
        int arr3[size1+size2];
        read_array(arr1,size1);
        read_array(arr2,size2);
        int new_size=marge_arr(arr1,arr2,size1,size2,arr3);
        bubble_sort(arr3,new_size);
        int median=find_median(arr3,new_size);
        printf("Median --> %i\n",median);
        return 0;
}
